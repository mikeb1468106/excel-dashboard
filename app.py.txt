import streamlit as st
import pandas as pd

# Load Excel file
file_path = 'Noah_grading_sheet_Analysis_updated (3).xlsx'
df = pd.read_excel(file_path)

st.set_page_config(page_title='Excel Dashboard', layout='wide')

st.title('Web-Based Dashboard')
st.markdown('Dashboard generated from uploaded Excel file')

# Sidebar filters
st.sidebar.header('Filters')

columns = df.select_dtypes(include=['object']).columns.tolist()
filters = {}
for col in columns:
    filters[col] = st.sidebar.multiselect(f'Select {col}', df[col].unique())

filtered_df = df.copy()
for col, selected in filters.items():
    if selected:
        filtered_df = filtered_df[filtered_df[col].isin(selected)]

# KPIs
st.subheader('Key Metrics')
num_cols = df.select_dtypes(include=['int64','float64']).columns
kpi_cols = st.columns(len(num_cols))
for i, col in enumerate(num_cols):
    kpi_cols[i].metric(label=col, value=round(filtered_df[col].mean(),2))

# Data table
st.subheader('Data Table')
st.dataframe(filtered_df, use_container_width=True)

# Charts
st.subheader('Charts')
for col in num_cols:
    st.bar_chart(filtered_df[col])
